idea sql tablas
-- 1. player_deaths (con lake_name)
CREATE TABLE IF NOT EXISTS player_deaths (
    id INT AUTO_INCREMENT PRIMARY KEY,
    player_id VARCHAR(100),
    death_cause VARCHAR(50),
    position_x FLOAT,
    position_y FLOAT,
    position_z FLOAT,
    zone_name VARCHAR(50),
    level_name VARCHAR(50),
    lake_name VARCHAR(100),
    timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 2. destructible_cubes (simplificada)
CREATE TABLE IF NOT EXISTS destructible_cubes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    cube_type VARCHAR(50),
    position_x FLOAT,
    position_y FLOAT,
    position_z FLOAT,
    zone_name VARCHAR(50),
    destruction_count INT DEFAULT 1,
    timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 3. enemy_stats (simplificada)
CREATE TABLE IF NOT EXISTS enemy_stats (
    id INT AUTO_INCREMENT PRIMARY KEY,
    enemy_type VARCHAR(50),
    time_to_kill FLOAT,
    damage_dealt FLOAT DEFAULT 0,
    timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
);

-- 4. acid_lakes (OPCIONAL - para tracking específico)
CREATE TABLE IF NOT EXISTS acid_lakes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    lake_name VARCHAR(100),
    position_x FLOAT,
    position_y FLOAT,
    position_z FLOAT,
    deaths_count INT DEFAULT 1,
    zone_name VARCHAR(50),
    last_updated DATETIME DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    UNIQUE KEY unique_lake (lake_name, position_x, position_z)
);

-- 5. tutorial_deaths (OPCIONAL - para despres)
CREATE TABLE IF NOT EXISTS tutorial_deaths (
    id INT AUTO_INCREMENT PRIMARY KEY,
    player_id VARCHAR(100),
    tutorial_phase VARCHAR(50),
    death_cause VARCHAR(50),
    completed BOOLEAN DEFAULT FALSE,
    attempts INT DEFAULT 1,
    timestamp DATETIME DEFAULT CURRENT_TIMESTAMP
);
